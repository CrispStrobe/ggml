cmake_minimum_required(VERSION 3.14...3.28)
project(ggml-wasm-q8-rounding-controls LANGUAGES C CXX)
if (NOT EMSCRIPTEN)
    message(FATAL_ERROR "These candidate controls require hosted Emscripten/WASM SIMD")
endif()
# Dedicated entry point avoids the ordinary test suite's host-ISA discovery.
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. ggml)
add_executable(test-wasm-q8-rounding EXCLUDE_FROM_ALL ../test-wasm-q8-rounding.cpp)
target_include_directories(test-wasm-q8-rounding PRIVATE ../../src ../../src/ggml-cpu)
target_compile_features(test-wasm-q8-rounding PRIVATE cxx_std_17)
target_compile_options(test-wasm-q8-rounding PRIVATE -msimd128)
target_link_libraries(test-wasm-q8-rounding PRIVATE ggml ggml-cpu)
